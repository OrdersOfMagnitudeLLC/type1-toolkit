cmake_minimum_required(VERSION 3.20)
project(nscomp CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

add_executable(nscomp_bench nscomp_bench.cpp nscomp.cpp)
target_include_directories(nscomp_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_library(ZSTD_LIB NAMES zstd libzstd)
find_library(LZ4_LIB NAMES lz4 liblz4)
if(ZSTD_LIB AND LZ4_LIB)
    target_link_libraries(nscomp_bench PRIVATE ${ZSTD_LIB} ${LZ4_LIB})
else()
    message(WARNING "zstd/lz4 not found")
endif()
target_compile_options(nscomp_bench PRIVATE -O3 -march=native)

add_executable(nscomp_server nscomp_server.cpp nscomp.cpp)
target_include_directories(nscomp_server PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_library(SSL_LIB NAMES ssl libssl)
find_library(CRYPTO_LIB NAMES crypto libcrypto)
if(SSL_LIB AND CRYPTO_LIB AND ZSTD_LIB AND LZ4_LIB)
    target_link_libraries(nscomp_server PRIVATE ${SSL_LIB} ${CRYPTO_LIB} ${ZSTD_LIB} ${LZ4_LIB})
endif()
target_compile_options(nscomp_server PRIVATE -O3 -march=native)
