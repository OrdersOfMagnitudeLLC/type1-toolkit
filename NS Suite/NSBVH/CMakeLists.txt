cmake_minimum_required(VERSION 3.20)
project(nsbvh CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)


# --- Highway: system package, vendored copy, or bare library ---
if(NOT TARGET hwy)
    find_package(HWY QUIET)
endif()
if(NOT TARGET hwy AND TARGET hwy::hwy)
    add_library(hwy INTERFACE)
    target_link_libraries(hwy INTERFACE hwy::hwy)
endif()
if(NOT TARGET hwy)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/highway/CMakeLists.txt")
        add_subdirectory(third_party/highway EXCLUDE_FROM_ALL)
    else()
        find_library(HWY_LIB NAMES hwy libhwy)
        find_path(HWY_INCLUDE_DIR NAMES hwy/highway.h)
        if(HWY_LIB)
            add_library(hwy UNKNOWN IMPORTED)
            set_target_properties(hwy PROPERTIES
                IMPORTED_LOCATION "${HWY_LIB}"
                INTERFACE_INCLUDE_DIRECTORIES "${HWY_INCLUDE_DIR}")
        else()
            message(WARNING "Highway (hwy) not found - using stub interface target")
            add_library(hwy INTERFACE)
        endif()
    endif()
endif()
if(NOT TARGET hwy_contrib)
    if(TARGET hwy::hwy_contrib)
        add_library(hwy_contrib INTERFACE)
        target_link_libraries(hwy_contrib INTERFACE hwy::hwy_contrib)
    else()
        find_library(HWY_CONTRIB_LIB NAMES hwy_contrib libhwy_contrib)
        if(HWY_CONTRIB_LIB)
            add_library(hwy_contrib UNKNOWN IMPORTED)
            set_target_properties(hwy_contrib PROPERTIES
                IMPORTED_LOCATION "${HWY_CONTRIB_LIB}")
        else()
            add_library(hwy_contrib INTERFACE)
        endif()
    endif()
endif()
# --- end Highway ---

add_executable(nsbvh nsbvh_bench.cpp)
target_include_directories(nsbvh PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
find_library(EMBREE_LIB NAMES embree4 embree3 embree)
if(EMBREE_LIB)
    target_link_libraries(nsbvh PRIVATE ${EMBREE_LIB} hwy)
else()
    message(WARNING "embree not found")
    target_link_libraries(nsbvh PRIVATE hwy)
endif()
target_compile_options(nsbvh PRIVATE -O3 -march=native)

add_executable(nsbvh_simple nsbvh_simple.cpp)
target_include_directories(nsbvh_simple PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../NSHash)
if(EMBREE_LIB)
    target_link_libraries(nsbvh_simple PRIVATE ${EMBREE_LIB} hwy hwy_contrib)
else()
    target_link_libraries(nsbvh_simple PRIVATE hwy hwy_contrib)
endif()
target_compile_options(nsbvh_simple PRIVATE -O3 -march=native)
