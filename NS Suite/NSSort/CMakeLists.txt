cmake_minimum_required(VERSION 3.20)
project(nssort CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)


# --- Highway: system package, vendored copy, or bare library ---
if(NOT TARGET hwy)
    find_package(HWY QUIET)
endif()
if(NOT TARGET hwy AND TARGET hwy::hwy)
    add_library(hwy INTERFACE)
    target_link_libraries(hwy INTERFACE hwy::hwy)
endif()
if(NOT TARGET hwy)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/highway/CMakeLists.txt")
        add_subdirectory(third_party/highway EXCLUDE_FROM_ALL)
    else()
        find_library(HWY_LIB NAMES hwy libhwy)
        find_path(HWY_INCLUDE_DIR NAMES hwy/highway.h)
        if(HWY_LIB)
            add_library(hwy UNKNOWN IMPORTED)
            set_target_properties(hwy PROPERTIES
                IMPORTED_LOCATION "${HWY_LIB}"
                INTERFACE_INCLUDE_DIRECTORIES "${HWY_INCLUDE_DIR}")
        else()
            message(WARNING "Highway (hwy) not found - using stub interface target")
            add_library(hwy INTERFACE)
        endif()
    endif()
endif()
if(NOT TARGET hwy_contrib)
    if(TARGET hwy::hwy_contrib)
        add_library(hwy_contrib INTERFACE)
        target_link_libraries(hwy_contrib INTERFACE hwy::hwy_contrib)
    else()
        find_library(HWY_CONTRIB_LIB NAMES hwy_contrib libhwy_contrib)
        if(HWY_CONTRIB_LIB)
            add_library(hwy_contrib UNKNOWN IMPORTED)
            set_target_properties(hwy_contrib PROPERTIES
                IMPORTED_LOCATION "${HWY_CONTRIB_LIB}")
        else()
            add_library(hwy_contrib INTERFACE)
        endif()
    endif()
endif()
# --- end Highway ---

add_executable(nssort NSSort.cpp nssort_bench.cpp)
target_include_directories(nssort PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/ips4o)
target_compile_definitions(nssort PRIVATE USE_IPS4O BENCH_MODE IPS4O_PARALLEL)
find_library(TBB_LIB NAMES tbb libtbb)
find_library(SODIUM_LIB NAMES sodium libsodium)
set(NSSORT_LIBS Threads::Threads OpenMP::OpenMP_CXX hwy atomic)
if(TBB_LIB)
    list(APPEND NSSORT_LIBS ${TBB_LIB})
endif()
if(SODIUM_LIB)
    list(APPEND NSSORT_LIBS ${SODIUM_LIB})
endif()
target_link_libraries(nssort PRIVATE ${NSSORT_LIBS})
target_compile_options(nssort PRIVATE -O3 -march=native -mavx2)
target_link_options(nssort PRIVATE -Wl,--allow-multiple-definition)

add_executable(bench_vqsort bench_vqsort.cpp)
target_compile_options(bench_vqsort PRIVATE -O3 -march=native)
target_link_libraries(bench_vqsort PRIVATE hwy hwy_contrib)
